def sha(path: Path) -> str:
    result = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            result.update(chunk)
    return result.hexdigest()


def digest(value: Any) -> str:
    text = value if isinstance(value, str) else json.dumps(value, sort_keys=True)
    return hashlib.sha256(text.encode()).hexdigest()


def read(path: Path, default: Any = None) -> Any:
    return json.loads(path.read_text()) if path.exists() else default


def rows(path: Path) -> list[dict[str, Any]]:
    return (
        [json.loads(line) for line in path.read_text().splitlines() if line.strip()]
        if path.exists()
        else []
    )


def close(left: float, right: float, label: str) -> None:
    if not math.isclose(left, right, rel_tol=1e-9, abs_tol=1e-9):
        raise ValueError(f"{label}: {left} != {right}")


def physical_account(
    cap: dict[str, Any], records: list[dict[str, Any]]
) -> tuple[dict[str, Any], list[dict[str, Any]]]:
    """Reconciliation resolves an existing attempt; it never adds a second request/cost."""
    groups: dict[str, list[dict[str, Any]]] = {}
    for record in records:
        groups.setdefault(record["attempt"], []).append(record)
    unique = []
    for attempt, group in groups.items():
        statuses = {r["status"] for r in group}
        if not statuses <= {"returned", "ambiguous_failure", "orphan_reconciled"}:
            raise ValueError(f"unsupported physical status for {attempt}: {statuses}")
        if "returned" in statuses and len(statuses) > 1:
            raise ValueError(f"request has contradictory returned/failed outcomes: {attempt}")
        if len({float(r["reserved_usd"]) for r in group}) != 1:
            raise ValueError(f"request reservation changed: {attempt}")
        if len({float(r["conservative_usd"]) for r in group if r["status"] == "returned"}) > 1:
            raise ValueError(f"returned request charge changed: {attempt}")
        chosen = next((r for r in group if r["status"] == "orphan_reconciled"), group[-1])
        unique.append(chosen)
    inflight = cap.get("inflight", {})
    if set(groups) & set(inflight):
        raise ValueError("a terminal physical request remains in flight")
    if len(groups) + len(inflight) != cap["attempts"]:
        raise ValueError("physical attempt identities do not reconcile with the cap")
    returned = sum(float(r["conservative_usd"]) for r in unique if r["status"] == "returned")
    uncertain = sum(float(r["reserved_usd"]) for r in unique if r["status"] != "returned")
    close(returned, float(cap["actual_usd"]), "returned conservative cost")
    close(uncertain, float(cap["uncertain_usd"]), "retained failed reservations")
    inflight_usd = sum(float(v) for v in inflight.values())
    result = {
        "returned_conservative_usd": returned,
        "retained_failed_reservation_usd": uncertain,
        "inflight_reservation_usd": inflight_usd,
        "committed_upper_usd": returned + uncertain + inflight_usd,
        "physical_attempts": cap["attempts"],
        "terminal_attempts": len(unique),
        "returned_requests": sum(r["status"] == "returned" for r in unique),
        "ambiguous_failures": sum(r["status"] == "ambiguous_failure" for r in unique),
        "orphan_reconciled_requests": sum(r["status"] == "orphan_reconciled" for r in unique),
        "physical_record_rows": len(records),
        "inflight_requests": len(inflight),
        "screening_cap_usd": 4.0,
        "whole_run_cap_usd": 29.0,
        "accounting_note": "Orphan reconciliation retains the existing request reservation; it is neither a new attempt nor an additional cost. Returned ledger USD and conservative committed USD are alternative accounting views, not additive charges.",
    }
    return result, unique


def screening_rows(
    screening: dict[str, Any], ledger: list[dict[str, Any]], physical: list[dict[str, Any]]
) -> list[dict[str, Any]]:
    result = []
    for key, row in sorted(screening["tasks"].items(), key=lambda item: int(item[0])):
        episodes = row["episodes"]
        valid = [e for e in episodes if e["is_valid"]]
        if (
            len(valid) != row["valid_episodes"]
            or sum(bool(e["behavioral_success"]) for e in valid) != row["valid_successes"]
        ):
            raise ValueError(f"screening counters disagree for {key}")
        if [e["execution_index"] for e in valid] != row["valid_episode_indices"]:
            raise ValueError(f"valid execution mapping differs for {key}")
        if row["status"] == "CONFIRMED_ZERO" and (len(valid) != 16 or row["valid_successes"] != 0):
            raise ValueError(f"invalid ZERO classification for {key}")
        taskledger = [
            r for r in ledger if str(r.get("task_id")) == key and r.get("arm") == "screen"
        ]
        taskphysical = [
            r
            for r in physical
            if str(r.get("attribution", {}).get("task_id")) == key and r["phase"] == "screening"
        ]
        result.append(
            {
                "task_id": int(key),
                "status": row["status"],
                "valid_episodes": row["valid_episodes"],
                "valid_successes": row["valid_successes"],
                "requested_executions": row["executions"],
                "returned_executions": len(episodes),
                "invalid_returned_executions": len(episodes) - len(valid),
                "executions_without_returned_trace": row["executions"] - len(episodes),
                "recovered_provider_errors": row["recovered_provider_errors"],
                "recovered_episode_executions": row["recovered_episode_executions"],
                "recovered_behavioral_results": len(row["recoveries"]),
                "provider_retry_events": sum(r.get("event") == "infra_retry" for r in taskledger),
                "terminal_ledger_infra_failures": sum(
                    r.get("event") == "infra_failure" for r in taskledger
                ),
                "physical_requests_with_terminal_record": len(taskphysical),
                "ledger_usd": sum(
                    float(r.get("usd") or 0) for r in taskledger if r.get("event") == "call"
                ),
                "retained_failed_reservation_usd": sum(
                    float(r["reserved_usd"]) for r in taskphysical if r["status"] != "returned"
                ),
                "infrastructure_reason": row.get("infrastructure_reason"),
                "valid_episode_execution_indices": row["valid_episode_indices"],
                "execution_evidence": [
                    {
                        k: e.get(k)
                        for k in (
                            "execution_index",
                            "target_valid_index",
                            "slot_attempt",
                            "is_valid",
                            "behavioral_success",
                            "episode_ids",
                            "trace_file",
                            "trace_sha256",
                            "file_sha256",
                        )
                    }
                    for e in episodes
                ],
            }
        )
    return result


def semantic_result(raw: dict[str, Any]) -> dict[str, Any]:
    result = raw["result"]
    return {
        "optimizer_call_index": raw["optimizer_call_index"],
        "doses": raw["doses"],
        "decision": result["decision"],
        "version": result.get("version"),
        "source_sha256": result["source_sha256"],
        "input_sha256": result["input_sha256"],
        "result_sha256": digest(result),
        "finding_count": len(result.get("findings", [])),
    }


def candidate_result(raw: dict[str, Any], semantics: list[dict[str, Any]]) -> dict[str, Any]:
    if digest(raw["source"]) != raw["source_sha256"]:
        raise ValueError("candidate source hash differs")
    structural = raw["structural"]
    privilege = raw["privilege"]
    identity = [x for x in privilege if x.startswith("DOSE = 0")]
    lexical = [
        x for x in privilege if not x.startswith("DOSE = 0") and "semantic_privilege_" not in x
    ]
    screened = [s for s in semantics if s["optimizer_call_index"] == raw["optimizer_call_index"]]
    semantic = (
        screened[-1]["decision"]
        if screened
        else "UNCERTAIN"
        if any("semantic_privilege_uncertain" in x for x in privilege)
        else "FAIL"
        if any("semantic_privilege_fail" in x for x in privilege)
        else "NOT_RUN"
    )
    return {
        **{
            k: raw.get(k)
            for k in (
                "candidate_id",
                "parent_candidate_id",
                "optimizer_call_index",
                "requested_operation",
                "source_sha256",
                "mechanism",
                "endpoint",
                "remaining_calls",
                "remaining_policy",
            )
        },
        "gates": {
            "structural": "FAIL" if structural else "PASS",
            "d0_identity": "NOT_RUN" if structural else "FAIL" if identity else "PASS",
            "lexical": "NOT_RUN" if structural or identity else "FAIL" if lexical else "PASS",
            "semantic": semantic,
            "solvability": {k: json.loads(raw["solvability"]).get(k) for k in ("ok", "source")}
            if raw.get("solvability")
            else "NOT_RUN",
        },
        "structural_reason_count": len(structural),
        "identity_reason_count": len(identity),
        "lexical_reason_count": len(lexical),
        "rejection_reason_sha256": digest(raw["rejection_reason"])
        if raw.get("rejection_reason")
        else None,
        "candidate_record_sha256": digest(raw),
        "semantic_records": screened,
    }


def adaptation_task(run: Path, task: int, row: dict[str, Any]) -> dict[str, Any]:
    original = row["original"]
    own = row.get("summary") or read(run / f"task-{task}/adaptation/summary.json", {})
    directory = run / f"task-{task}/adaptation"
    semantics = [semantic_result(r) for r in rows(run / f"task-{task}/semantic_gates.jsonl")]
    candidates = [candidate_result(c, semantics) for c in own.get("candidates", [])]
    inputs = rows(directory / "designer_inputs.jsonl")
    feedback = []
    for record in rows(directory / "feedback.jsonl"):
        raw = record["feedback"]
        feedback.append(
            {
                "to_call_index": record["call_index"],
                **{
                    k: raw.get(k)
                    for k in (
                        "operation",
                        "category",
                        "candidate_id",
                        "source_sha256",
                        "remaining_calls",
                        "remaining_policy",
                        "endpoint",
                        "failed_episode_id",
                    )
                },
                "reason_sha256": digest(raw.get("reason", "")),
                "feedback_sha256": digest(raw),
            }
        )
    slots = []
    for index in (1, 2, 3):
        candidate = next((c for c in candidates if c["optimizer_call_index"] == index), None)
        dispatched = next((r for r in inputs if r["call_index"] == index), None)
        if candidate:
            slots.append(
                {
                    "slot": f"C{index}",
                    "status": "PRODUCED",
                    "candidate_id": candidate["candidate_id"],
                }
            )
        elif dispatched:
            slots.append(
                {
                    "slot": f"C{index}",
                    "status": "DISPATCHED_WITHOUT_CANDIDATE",
                    "request_sha256": dispatched["request_sha256"],
                }
            )
        else:
            reason = (
                "first viable family already frozen"
                if own.get("first_viable_family_index")
                else "reference unavailable"
                if not original["reference_ok"]
                else "semantic UNCERTAIN stopped DESIGN"
                if any(s["decision"] == "UNCERTAIN" for s in semantics)
                else "task/global termination or budget prevented call"
            )
            slots.append(
                {
                    "slot": f"C{index}",
                    "status": "NOT_NEEDED" if own.get("first_viable_family_index") else "NOT_RUN",
                    "reason": reason,
                }
            )
    frozen = own.get("frozen_family")
    freeze = (
        None
        if not frozen
        else {
            k: frozen[k]
            for k in ("candidate_id", "source_sha256", "optimizer_call_index", "endpoint")
        }
    )
    confirmation = own.get("confirmation") or read(run / f"task-{task}/confirmation.json")
    if own.get("designer_calls", len(inputs)) > 3 or own.get("adaptation_rollouts", 0) > 30:
        raise ValueError("reported task exceeded frozen designer/adaptation allocation")
    return {
        "task_id": task,
        "task_status": row["status"],
        "design_status": own.get("status", "NOT_RUN"),
        "reference": {
            k: original.get(k)
            for k in (
                "reference_status",
                "reference_ok",
                "reference_id",
                "reference_sha256",
                "reference_steps",
            )
        },
        "evidence": {
            k: original.get(k)
            for k in (
                "all_original_episode_ids",
                "all_original_trace_sha256",
                "original_zero_evidence_sha256",
                "evidence_n",
                "selection_seed",
                "evidence_ids",
                "selected_trace_sha256",
                "evidence_sha256",
                "designer_evidence_sha256",
                "valid_episode_execution_indices",
            )
        },
        "designer_slots": slots,
        "candidates": candidates,
        "typed_feedback": feedback,
        "designer_inputs": inputs,
        "semantic_gates": semantics,
        "frozen_family": freeze,
        "control": own.get("control")
        or {
            "status": "NOT_NEEDED"
            if frozen and frozen["endpoint"]["verdict"] == "in_band"
            else "NOT_RUN"
        },
        "search_accepted": bool(own.get("search_accepted")),
        "accepted_dose": own.get("accepted_dose"),
        "search_successes": own.get("search_successes"),
        "search_n": own.get("search_n"),
        "final_environment_hash": own.get("final_environment_hash"),
        "confirmation": confirmation,
        "designer_calls": own.get("designer_calls", len(inputs)),
        "adaptation_rollouts": own.get("adaptation_rollouts", 0),
        "adaptation_requested_batches": rows(directory / "physical_batches.jsonl"),
        "confirmation_requested_batches": rows(run / f"task-{task}/confirmation_batches.jsonl"),
        "adaptation_budget": read(directory / "rollout_budget.json", []),
        "measurements": rows(directory / "measurements.jsonl"),
        "interruption_kind": (own.get("interruption") or {}).get("kind"),
        "completed_without_infrastructure_failure": own.get(
            "completed_without_infrastructure_failure", False
        ),
    }

HISTORICAL_COMMITTED_USD = 7.994786072


def prior_core(repo: Path, frozen: Path) -> dict[str, Any]:
    path = frozen / "core_manifest.json"
    manifest = read(path)
    if not manifest or [r["task_id"] for r in manifest["tasks"]] != CORE:
        raise ValueError("frozen prior three-task core is missing or changed")
    evidence_path = repo / manifest["prior_qualified_evidence_path"]
    prior = read(evidence_path)
    screening_path = repo / manifest["prior_screening_path"]
    if (
        sha(evidence_path) != manifest["prior_qualified_evidence_sha256"]
        or prior["tasks"] != manifest["tasks"]
        or prior["qualified_task_ids"] != CORE
        or sha(screening_path) != manifest["prior_screening_sha256"]
        or prior["screening_sha256"] != manifest["prior_screening_sha256"]
    ):
        raise ValueError("historical core provenance differs")
    for row in manifest["tasks"]:
        if (
            row["evidence_n"] != 16
            or len(row["all_original_episode_ids"]) != 16
            or len(row["all_original_trace_sha256"]) != 16
            or len(row["evidence_ids"]) != 3
            or row["selection_seed"] != row["task_id"]
        ):
            raise ValueError("historical sixteen/three evidence shape differs")
    for relative, expected in manifest["raw_file_hashes"].items():
        if sha(repo / relative) != expected:
            raise ValueError("historical original trace bytes changed: " + relative)
    previous_result = (
        repo / "experiments/alfworld_e6/results/iterative_low_viability_v2/result.json"
    )
    historical = read(previous_result)["cost"]
    close(
        float(historical["committed_upper_usd"]),
        HISTORICAL_COMMITTED_USD,
        "historical V2 committed cost",
    )
    return {
        "status": "LOCAL_HASH_BINDINGS_VERIFIED",
        "task_ids": CORE,
        "prior_commit": manifest["prior_commit"],
        "core_manifest_sha256": sha(path),
        "prior_qualified_evidence_sha256": sha(evidence_path),
        "prior_screening_sha256": sha(screening_path),
        "verified_original_files": len(manifest["raw_file_hashes"]),
        "tasks": manifest["tasks"],
        "historical_sunk_cost": {
            "committed_usd": historical["committed_upper_usd"],
            "ledger_usd": historical["actual_ledger_usd"],
            "included_in_continuation_cap": False,
            "source_sha256": sha(previous_result),
        },
    }


def phase_costs(
    cap: dict[str, Any], physical: list[dict[str, Any]], ledger: list[dict[str, Any]], run: Path
) -> dict[str, Any]:
    cost, unique = physical_account(cap, physical)
    calls = [r for r in ledger if r.get("event") == "call"]
    cost.update(
        adaptation_confirmation_cap_usd=25.0,
        screening_cap_usd=4.0,
        whole_run_cap_usd=29.0,
        actual_ledger_usd=sum(float(r.get("usd") or 0) for r in calls),
        historical_sunk_cost_included=False,
    )
    groups = {
        name: {
            "returned_conservative_usd": 0.0,
            "retained_failed_usd": 0.0,
            "physical_requests": 0,
            "ledger_usd": 0.0,
        }
        for name in ("topup", "designer", "adaptation_policy", "confirmation", "other")
    }

    def category(row: dict[str, Any], physical_row: bool) -> str:
        attr = row.get("attribution", {}) if physical_row else row
        if row.get("phase") == "screening" if physical_row else attr.get("arm") == "screen":
            return "topup"
        if attr.get("budget") == "designer":
            return "designer"
        if attr.get("phase") == "confirm":
            return "confirmation"
        return "adaptation_policy" if attr.get("budget") == "search" else "other"

    for row in unique:
        group = groups[category(row, True)]
        group["physical_requests"] += 1
        if row["status"] == "returned":
            group["returned_conservative_usd"] += float(row["conservative_usd"])
        else:
            group["retained_failed_usd"] += float(row["reserved_usd"])
    for row in calls:
        groups[category(row, False)]["ledger_usd"] += float(row.get("usd") or 0)
    for group in groups.values():
        group["terminal_committed_usd"] = (
            group["returned_conservative_usd"] + group["retained_failed_usd"]
        )
    baseline_path = run / "topup_cost_baseline.json"
    baseline = read(baseline_path)
    topup = float(groups["topup"]["terminal_committed_usd"])
    if baseline:
        close(float(baseline["screening_committed_usd"]), topup, "frozen top-up cost baseline")
        if sha(baseline_path) != cap.get("screening_baseline_sha256"):
            raise ValueError("top-up baseline hash differs from the shared cap")
        if float(cap["actual_usd"]) < float(baseline["actual_usd"]) or float(
            cap["uncertain_usd"]
        ) < float(baseline["uncertain_usd"]):
            raise ValueError("historical top-up reservation was refunded")
    cost["by_phase"] = groups
    cost["topup_cost_baseline"] = baseline
    cost["topup_baseline_sha256"] = sha(baseline_path) if baseline else None
    cost["topup_committed_usd"] = topup + (
        cost["inflight_reservation_usd"] if cap["phase"] == "screening" else 0
    )
    cost["adaptation_confirmation_committed_usd"] = (
        cost["committed_upper_usd"] - topup if baseline else 0.0
    )
    cost["effective_cumulative_ceiling_usd"] = (
        min(29.0, topup + 25.0) if baseline and baseline["screening_status"] == "qualified" else 4.0
    )
    cost["unused_topup_headroom_transferable"] = False
    cost["transport_stop"] = cap.get("stopped")
    return cost


def expected_decision(
    results: dict[str, Any] | None,
    screen: dict[str, Any],
    cap: dict[str, Any],
    error: dict[str, Any] | None,
    correctness: dict[str, Any],
) -> str:
    defect = (
        correctness.get("decision") == "IMPLEMENTATION_FAILURE"
        or cap.get("bound_violation")
        or (error and error.get("kind") in CORRECTNESS)
    )
    if results is None:
        return (
            "IMPLEMENTATION_FAILURE"
            if defect or screen["status"] == "IMPLEMENTATION_FAILURE"
            else "INCONCLUSIVE"
            if screen["status"] == "qualified"
            else "TOPUP_INCONCLUSIVE"
        )
    if defect:
        return "IMPLEMENTATION_FAILURE"
    rows_ = list(results["tasks"].values())
    summaries = [r["summary"] for r in rows_ if r.get("summary")]
    confirmed = sum(bool(s.get("confirmation", {}).get("in_band_l")) for s in summaries)
    failed_confirmations = sum(
        bool(s.get("search_accepted"))
        and s.get("confirmation", {}).get("n") == 16
        and s["confirmation"].get("in_band_l") is False
        for s in summaries
    )
    completed = sum(r["status"] == "completed" for r in rows_)
    referenced = sum(r["original"]["reference_ok"] for r in rows_)
    if confirmed >= 2:
        return "LOW_VIABLE"
    if error or completed < referenced:
        return "INCONCLUSIVE"
    if confirmed == 1 or failed_confirmations >= 2:
        return "LOW_PARTIAL_SIGNAL"
    return "LOW_NOT_WORKING" if completed >= 3 else "INCONCLUSIVE"


def reference_inventory(run: Path, ids: list[int]) -> list[dict[str, Any]]:
    result = []
    for task in ids:
        path = run / f"task-{task}/privileged_reference.json"
        raw = read(path)
        result.append(
            {
                "task_id": task,
                "status": "NOT_RUN_OR_NOT_COMPLETED"
                if raw is None
                else "REFERENCE_AVAILABLE"
                if raw["success"]
                else "REFERENCE_UNAVAILABLE",
                "reference_id": raw.get("reference_id") if raw else None,
                "reference_sha256": digest(raw) if raw else None,
                "reference_file_sha256": sha(path) if raw else None,
            }
        )
    return result

            f"| {row['task_id']} | {row['valid_successes']}/{row['valid_episodes']} | {row['status']} | {row['invalid_returned_executions']} / {row['executions_without_returned_trace']} | {row['recovered_provider_errors']} / {row['recovered_episode_executions']} | {row['ledger_usd']:.9f} |"
        )
    lines += [
        "",
        "Tasks150–162 remain consumed;162 was never resumed. Prior screening reports remain unchanged.",
        "",
        "## Cost",
        "",
        f"Historical V2 committed cost **${result['prior_core']['historical_sunk_cost']['committed_usd']:.9f}** is excluded from this continuation.",
        "",
        "| Phase | Ledger USD | Conservative returned USD | Retained failed USD | Physical requests |",
        "| --- | ---: | ---: | ---: | ---: |",
    ]
    for name, row in cost["by_phase"].items():
        lines.append(
            f"| {name} | {row['ledger_usd']:.9f} | {row['returned_conservative_usd']:.9f} | {row['retained_failed_usd']:.9f} | {row['physical_requests']} |"
        )
    lines += [
        "",
        f"Top-up committed **${cost['topup_committed_usd']:.9f}/$4**; adaptation+confirmation committed **${cost['adaptation_confirmation_committed_usd']:.9f}/$25**. Total continuation committed **${cost['committed_upper_usd']:.9f}**; unresolved in-flight reservations ${cost['inflight_reservation_usd']:.9f}.",
        "The immutable top-up baseline is retained. Unused top-up headroom cannot enlarge adaptation. Orphan reconciliation is counted once; ledger and conservative costs are alternative accounting views.",
        "",
        "## References and method execution",
        "",
    ]
    for row in result["references"]:
        lines.append(
            f"- Task{row['task_id']}: {row['status']}; reference ID `{row['reference_id']}`; exact-record SHA `{row['reference_sha256']}`."
        )
    if result["stage_interruption"]:
        error = result["stage_interruption"]["error"]
        lines += [
            f"- Terminal stage interruption: {result['stage_interruption']['stage']} / {error['kind']} / {error['reason']}."
        ]
    if not result["adaptation_tasks"]:
        lines += [
            "C1/C2/C3, admission gates, solvability, endpoint, CONTROL, search acceptance and K16: **NOT_RUN / N/A**."
        ]
        lines += [
            "Primary efficacy result: **N/A; no four-task sample formed**."
            if primary is None
            else "The four-task denominator is preserved, but LOW was not evaluated before the interruption; no efficacy conclusion follows."
        ]
    else:
        lines += [
            f"Primary: **{primary['K16_B_L']}/4 K16-confirmed B_L**; B_T {primary['K16_B_T']}/4; search accepted {primary['search_accepted']}/4. Historical bands: B_L4–12/16, B_T7–9/16."
        ]
        for task in result["adaptation_tasks"]:
            lines += [
                "",
                f"### Task{task['task_id']}",
                "",
                f"Task status {task['task_status']}; DESIGN {task['design_status']}; evidence SHA `{task['evidence']['evidence_sha256']}`.",
            ]
            for slot in task["designer_slots"]:
                lines.append(
                    f"- {slot['slot']}: {slot['status']}; {slot.get('candidate_id', slot.get('reason', 'request interrupted'))}."
                )
            for candidate in task["candidates"]:
                lines += [
                    f"- C{candidate['optimizer_call_index']}: {candidate['mechanism']}; parent {candidate['parent_candidate_id']}; {candidate['requested_operation']}; source `{candidate['source_sha256']}`.",
                    f"  Gates `{json.dumps(candidate['gates'], sort_keys=True)}`; endpoint `{candidate['endpoint']}`.",
                ]
            for feedback in task["typed_feedback"]:
                lines.append(
                    f"- Feedback to C{feedback['to_call_index']}: {feedback['operation']}/{feedback['category']}; exact private packet SHA `{feedback['feedback_sha256']}`."
                )
            confirmation = task["confirmation"]
            lines += [
                f"- Frozen family: `{json.dumps(task['frozen_family'], sort_keys=True)}`.",
                f"- CONTROL: `{json.dumps(task['control'], sort_keys=True)}`.",
                f"- Search accepted {task['search_accepted']}, s/n {task['search_successes']}/{task['search_n']}, dose {task['accepted_dose']}.",
                f"- K16: {'NOT_RUN; not a failed confirmation' if not confirmation else str(confirmation['successes']) + '/16, B_L=' + str(confirmation['in_band_l']) + ', B_T=' + str(confirmation['in_band_t'])}.",
                f"- Designer calls {task['designer_calls']}/3; adaptation policy episodes {task['adaptation_rollouts']}/30.",
            ]
    lines += [
        "",
        "## Decision and next action",
        "",
        result["recommendation"],
        "",
        f"Single direct blocker: {result['review'].get('single_blocker', 'Final independent outcome review pending.')}",
        "",
        "Raw traces, source, prompts, references and detailed feedback/gate bodies remain local. Compact artifact manifests bind their bytes. K16 is evaluation-only; unperformed confirmations are not misses.",
        "",
    ]
    return "\n".join(lines)


def build(repo: Path, output: Path, review_path: Path | None = None) -> dict[str, Any]:
    if not output.resolve().is_relative_to(Path("/tmp")):
        raise ValueError("report drafts may only be written under /tmp")
    run = repo / "runs" / NAME
    frozen = repo / "experiments/alfworld_e6/frozen/iterative_low_viability_v3"
    screen, cap, summary, results = [
        read(run / name) for name in ("screening.json", "cap.json", "summary.json", "results.json")
    ]
    stage_failure = read(run / "stage_interruption.json")
    if not screen or not cap or not summary or screen["status"] == "running":
        raise ValueError("terminal driver screen/cap/summary artifacts required")
    if results is None and screen["status"] == "qualified" and stage_failure is None:
        raise ValueError("qualified top-up awaits adaptation; cannot finalize a live continuation")
    if screen["status"] not in ("qualified", "TOPUP_INCONCLUSIVE", "IMPLEMENTATION_FAILURE"):
        raise ValueError("unexpected frozen top-up status")
    core = prior_core(repo, frozen)
    if (
        screen["core_tasks"] != CORE
        or screen["core_manifest_sha256"] != core["core_manifest_sha256"]
    ):
        raise ValueError("top-up did not bind the exact prior core")
    audit = read(frozen / "used_task_audit.json")
    ids = sorted(int(key) for key in screen["tasks"])
    if (
        ids != audit["never_used_ids"][: len(ids)]
        or len(ids) > 10
        or any(task <= 162 for task in ids)
    ):
        raise ValueError("top-up used consumed, out-of-order or excess IDs")
    z4 = screen["qualified"][0] if len(screen["qualified"]) == 1 else None
    if len(screen["qualified"]) > 1 or (screen["status"] == "qualified" and z4 is None):
        raise ValueError("top-up must supply exactly one new ZERO")
    task_set = [*CORE, z4] if z4 is not None else []
    error = (
        (results or {}).get("interruption")
        or ((stage_failure or {}).get("error"))
        or screen.get("interruption")
    )
    if cap.get("inflight") and not error:
        raise ValueError("unresolved physical requests lack a terminal interruption")
    if results:
        pending = [
            r
            for r in results["tasks"].values()
            if r["status"] not in ("completed", "reference_unavailable")
        ]
        if pending and not error:
            raise ValueError("adaptation is still running")
        if [int(key) for key in results["tasks"]] != task_set:
            raise ValueError("adaptation task set/order differs from frozen core plus Z4")
    decision = expected_decision(
        results, screen, cap, error, read(run / "correctness_audit.json", {})
    )
    if (
        summary["decision"] != decision
        or summary.get("final_viability_decision", decision) != decision
    ):
        raise ValueError("driver summary decision differs from exact V3 terminal evidence")
    adaptations = (
        [adaptation_task(run, int(task), row) for task, row in results["tasks"].items()]
        if results
        else []
    )
    for task in adaptations:
        confirmation = task["confirmation"]
        if confirmation and (
            confirmation["n"] != 16
            or confirmation["in_band_l"] != (4 <= confirmation["successes"] <= 12)
            or confirmation["in_band_t"] != (7 <= confirmation["successes"] <= 9)
        ):
            raise ValueError("confirmation shape or historical band classification differs")
    physical = rows(run / "cap.attempts.jsonl")
    ledger = [r for path in sorted(run.glob("ledger.*.jsonl")) for r in rows(path)]
    cost = phase_costs(cap, physical, ledger, run)
    screen_rows = screening_rows(screen, ledger, physical_account(cap, physical)[1])
    if z4 is not None and not any(
        r["task_id"] == z4
        and r["status"] == "CONFIRMED_ZERO"
        and r["valid_episodes"] == 16
        and r["valid_successes"] == 0
        for r in screen_rows
    ):
        raise ValueError("Z4 is not a valid 0/16 ZERO")
    review = read(review_path, {}) if review_path else {}
    review.setdefault("status", "FINAL_AUDIT_PENDING")
    if review.get("decision") and review["decision"] != decision:
        raise ValueError("final review and frozen driver decision disagree")
    primary = (
        None
        if not task_set
        else {
            "denominator": 4,
            "LOW_evaluated": any(task["designer_calls"] > 0 for task in adaptations),
            "K16_B_L": sum(bool((r["confirmation"] or {}).get("in_band_l")) for r in adaptations),
            "K16_B_T": sum(bool((r["confirmation"] or {}).get("in_band_t")) for r in adaptations),
            "search_accepted": sum(r["search_accepted"] for r in adaptations),
            "completed_K16": sum(r["confirmation"] is not None for r in adaptations),
        }
    )
    if results and any(primary[key] != summary[key] for key in ("K16_B_L", "K16_B_T")):
        raise ValueError("primary counts differ from frozen driver summary")
    if not task_set:
        interpretation = "Top-up did not produce Z4. The preserved three-task core remains valid, but the four-task viability sample did not form; LOW efficacy was not evaluated."
        recommendation = "Stop. No automatic expansion or LOW change follows from top-up failure."
    elif not adaptations:
        interpretation = "The four-task set formed, but a reference/freeze boundary interruption stopped the run before LOW was evaluated. All four remain identified; no efficacy conclusion follows."
        recommendation = "Stop. Preserve the exact stage interruption and completed reference artifacts; no regeneration or automatic retry."
    else:
        interpretation = "The primary denominator remains the four frozen ZERO tasks, including unavailable references. Actual completed K16 outcomes determine confirmation evidence; endpoint success alone does not imply a failed confirmation."
        recommendation = (
            "FREEZE LOW. Stop LOW-only R&D and integrate complete AEA after review."
            if decision == "LOW_VIABLE"
            else review.get(
                "recommendation",
                "Stop. Final outcome review must identify the single direct functionality blocker and any next authorized change.",
            )
        )
    result = {
        "schema": "iterative-low-viability-v3-compact-report-v1",
        "final_decision": decision,
        "prior_core": core,
        "topup_freeze": screen["freeze"],
        "adaptation_freeze": results.get("freeze", {}) if results else {},
        "preflight": read(frozen / "preflight.json", read(frozen / "validation.json", {})),
        "topup": {
            "status": screen["status"],
            "stop_reason": screen.get("stop_reason") or error or "ten IDs exhausted"
            if screen["status"] == "TOPUP_INCONCLUSIVE"
            else screen.get("stop_reason") or error,
            "starting_fresh_id": audit["smallest_never_used"],
            "inspected_ids": ids,
            "Z4": z4,
            "tasks": screen_rows,
            "requested_executions": sum(r["requested_executions"] for r in screen_rows),
        },
        "final_task_set": task_set,
        "references": reference_inventory(run, task_set),
        "stage_interruption": stage_failure,
        "input_manifest": read(run / "input_manifest.json"),
        "input_manifest_sha256": sha(run / "input_manifest.json")
        if (run / "input_manifest.json").exists()
        else None,
        "primary": primary,
        "adaptation_tasks": adaptations,
        "candidate_count": sum(len(r["candidates"]) for r in adaptations),
        "designer_calls": sum(r["designer_calls"] for r in adaptations),
        "adaptation_policy_rollouts": sum(r["adaptation_rollouts"] for r in adaptations),
        "confirmation_requested_episodes": sum(
            sum(b["n"] for b in r["confirmation_requested_batches"]) for r in adaptations
        ),
        "cost": cost,
        "review": review,
        "interpretation": interpretation,
        "recommendation": recommendation,
        "report_helper_operations": {
            "API_calls": 0,
            "environment_resets": 0,
            "reference_generation": 0,
            "run_writes": 0,
            "research_writes": 0,
        },
    }
    inventory = [
        p
        for p in run.rglob("*")
        if p.is_file() and p.suffix != ".lock" and not p.name.endswith(".tmp")
    ]
    inventory += [p for p in frozen.rglob("*") if p.is_file()]
    inventory += [
        p
        for p in (
            repo / "scripts/e6_iterative_low_viability_v3.py",
            repo / "experiments/alfworld_e6/PREREG_ITERATIVE_LOW_TOPUP.md",
            repo / "experiments/alfworld_e6/PREREG_ITERATIVE_LOW_VIABILITY_V3.md",
        )
        if p.exists()
    ]
    manifest = {
        "helper_sha256": sha(Path(__file__)),
        "files": [
            {"path": str(p.relative_to(repo)), "bytes": p.stat().st_size, "sha256": sha(p)}
            for p in sorted(set(inventory))
        ],
    }
    if [
        read(run / name) for name in ("screening.json", "cap.json", "summary.json", "results.json")
    ] != [screen, cap, summary, results] or read(run / "stage_interruption.json") != stage_failure:
        raise ValueError("run artifacts changed during report generation")
    output.mkdir(parents=True, exist_ok=True)
    for name, value in (
        ("result.json", result),
        ("adaptation_tasks.json", adaptations),
        ("artifact_manifest.json", manifest),
    ):
        (output / name).write_text(json.dumps(value, indent=2, sort_keys=True) + "\n")
    (output / "report.md").write_text(render_v3(result))
    compact = {
        "files": [
            {"path": name, "bytes": (output / name).stat().st_size, "sha256": sha(output / name)}
            for name in (
                "result.json",
                "adaptation_tasks.json",
                "artifact_manifest.json",
                "report.md",
            )
        ]
    }
    (output / "compact_manifest.json").write_text(
        json.dumps(compact, indent=2, sort_keys=True) + "\n"
    )
    return result


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo", type=Path, default=Path("/home/kree/work/EnvJudge-aea-llm"))
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--review", type=Path)
    args = parser.parse_args()
    result = build(args.repo, args.output, args.review)
    print(
        json.dumps(
            {
                "decision": result["final_decision"],
                "topup": result["topup"]["status"],
                "Z4": result["topup"]["Z4"],
                "output": str(args.output),
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    main()
