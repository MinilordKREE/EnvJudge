    else: baselines[key]=p['baseline']

    changed=[k for k in p['baseline'] if p['baseline'][k]!=p['transformed'][k]]

    counts=changes_by_dose.setdefault(dose,Counter()); counts.update(changed); counts['probes']+=1;counts['identity']+=not changed

        focus_examples.append({'episode_id':eid,'step':step,'dose':dose,'promoted_commands':[c for c in p['transformed']['admissible_commands'] if p['transformed']['admissible_commands'].index(c)<p['baseline']['admissible_commands'].index(c)]})
