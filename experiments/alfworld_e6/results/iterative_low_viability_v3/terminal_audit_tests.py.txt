def write(path, value, *, jsonl=False):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        "".join(json.dumps(r) + "\n" for r in value) if jsonl else json.dumps(value)
    )


def rewrite(path, fn, *, jsonl=False):
    raw = (
        [json.loads(r) for r in path.read_text().splitlines()]
        if jsonl
        else json.loads(path.read_text())
    )
    fn(raw)
    write(path, raw, jsonl=jsonl)


def physical(status="returned", attempt="a"):
    row = {
        "attempt": attempt,
        "model": "qwen/qwen3-8b",
        "phase": "screening",
        "attribution": {"task_id": "163"},
        "reserved_usd": 0.5,
        "status": status,
    }
    if status == "returned":
        row["conservative_usd"] = 0.1
    return row


def cap(actual=0.1, uncertain=0, attempts=1, inflight=None):
    return {
        "actual_usd": actual,
        "uncertain_usd": uncertain,
        "attempts": attempts,
        "inflight": inflight or {},
    }


def test_orphan_reconciliation_keeps_one_attempt_and_full_failed_reserve():
    unique, result = audit.physical_costs(
        cap(0, 0.5), [physical("ambiguous_failure"), physical("orphan_reconciled")]
    )
    assert len(unique) == 1 and result["committed_usd"] == 0.5
    assert result["physical_attempts"] == result["orphan_reconciled_attempts"] == 1


@pytest.mark.parametrize(
    "rows,state",
    [
        ([physical("ambiguous_failure")], cap(0, 0)),
        ([physical(), physical("orphan_reconciled")], cap()),
        ([physical()], cap(attempts=2)),
        ([physical()], cap(inflight={"a": 0.5})),
    ],
)
def test_refunded_ambiguous_or_double_counted_attempts_rejected(rows, state):
    with pytest.raises(ValueError):
        audit.physical_costs(state, rows)


def test_unresolved_inflight_reservation_stays_in_committed_cost():
    unique, result = audit.physical_costs(
        cap(attempts=2, inflight={"b": 0.7}), [physical()]
    )
    assert len(unique) == 1 and result["committed_usd"] == pytest.approx(0.8)

@pytest.mark.parametrize(
    "mutation, expected",
    [
        ("gate_dose", "each_policy_batch_exact_admitted_source_dose"),
        ("gate_uncertain", "each_policy_batch_exact_admitted_source_dose"),
        ("runtime_source", "runtime_trace_exact_source_dose_seed"),
        ("reused_episode", "fresh_global_episode_identity"),
        ("k16_into_request", "exact_designer_messages_no_confirmation_input"),
        ("refunded_episode", "requested_episode_budget_never_refunded"),
        ("k16_changed_source", "confirmation_unchanged_runtime_environment"),
        ("k16_into_control", "confirmation_never_mutates_design_control_summary"),
    ],
)
def test_meaningful_provenance_tampering_is_detected(fixture, mutation, expected):
    obj, _, outer, _, _, directory = fixture
    if mutation == "gate_dose":
        rewrite(
            directory.parent / "semantic_gates.jsonl",
            lambda rows: rows[0]["doses"].remove(0.5),
            jsonl=True,
        )
    elif mutation == "gate_uncertain":
        rewrite(
            directory.parent / "semantic_gates.jsonl",
            lambda rows: rows[0]["result"].update(
                decision="UNCERTAIN", findings=[{"decision": "UNCERTAIN"}]
            ),
            jsonl=True,
        )
    elif mutation == "runtime_source":
        rewrite(
            directory / "traces.jsonl",
            lambda rows: rows[0]["candidate"].update(rules_code="altered"),
            jsonl=True,
        )
    elif mutation == "reused_episode":
        rewrite(
            directory / "traces.jsonl",
            lambda rows: rows[0].update(episode_id="original-0"),
            jsonl=True,
        )
    elif mutation == "k16_into_request":
        rewrite(
            directory / "privileged_designer_requests.jsonl",
            lambda rows: rows[0]["messages"][1].update(
                content=rows[0]["messages"][1]["content"] + " K16=3/16"
            ),
            jsonl=True,
        )
    elif mutation == "refunded_episode":
        rewrite(
            directory / "rollout_budget.json",
            lambda rows: rows[0].update(n=3, refunded=1),
        )
    elif mutation == "k16_changed_source":
        rewrite(
            directory.parent / "confirmation_traces.json",
            lambda rows: rows[0]["candidate"].update(
                rules_code="changed in confirmation"
            ),
        )
    elif mutation == "k16_into_control":
        outer["summary"]["control"]["status"] = "changed by confirmation"
    assert expected in evaluate_fixture(fixture)


def test_missing_or_stale_source_and_unknown_dose_are_not_admitted():
    source = "source"
    gate = {
        "result": {
            "source_sha256": audit.digest(source),
            "decision": "PASS",
            "findings": [{"decision": "PASS"}],
        },
        "doses": audit.DOSES[:],
    }
    assert audit.check_admission([gate], source, 0.5)
    assert not audit.check_admission([gate], "new source", 0.5)
    assert not audit.check_admission([gate], source, 0.3)
    assert not audit.check_admission([gate, copy.deepcopy(gate)], source, 0.5)
