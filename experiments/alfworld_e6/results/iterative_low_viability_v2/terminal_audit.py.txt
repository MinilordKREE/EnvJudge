#!/usr/bin/env python3
"""Independent, read-only final cap/closure/qualified-evidence cross-check."""
import ast
import collections
import random
import datetime
import hashlib
import json
from decimal import Decimal
from pathlib import Path
import sys

ROOT = Path('/home/kree/work/EnvJudge-aea-llm')
RUN = ROOT / 'runs/e6-iterative-low-viability-v2'
OUTPUT = Path('/tmp/viability_v2_terminal_supplement_corrected_final.json')
D = Decimal
failures, sources = [], {}
checks = 0


def read(path):
    data = path.read_bytes()
    sources[str(path.relative_to(ROOT)) if path.is_relative_to(ROOT) else str(path)] = hashlib.sha256(data).hexdigest()
    return data


def load(path):
    return json.loads(read(path), parse_float=D)


def rows(path):
    return [json.loads(line, parse_float=D) for line in read(path).splitlines() if line]


def check(ok, label, detail=None):
    global checks
    checks += 1
    if not ok:
        failures.append({'check': label, 'detail': detail})


def timestamp(row):
    return D(str(datetime.datetime.fromisoformat(row['ts'].replace('Z', '+00:00')).timestamp()))


screen = load(RUN / 'screening.json')
cap = load(RUN / 'cap.json')
stop = load(RUN / 'STOP.json')
attempts = rows(RUN / 'cap.attempts.jsonl')
ledger = [row for path in sorted(RUN.glob('ledger.*.jsonl')) for row in rows(path)]
ops = {p.name: load(p) for p in (RUN / 'operations').glob('*.json')}
check(screen['status'] == 'SCREENING_COST_LIMIT', 'terminal_screening_cost_limit')
check(cap['stopped'] == 'screening_cost_cap' and stop['kind'] == 'smoke_cost_cap', 'sticky_cap_stop')
check(not cap['inflight'], 'zero_inflight_reservations')
check(collections.Counter(op['status'] for op in ops.values()) == {'completed': 70, 'interrupted': 1}, 'all_71_operations_closed')
check(len(attempts) == cap['attempts'] == 3499, 'physical_attempt_count_exact')
check(len({a['attempt'] for a in attempts}) == len(attempts), 'physical_attempt_ids_unique')
returned = [a for a in attempts if a['status'] == 'returned']
failed = [a for a in attempts if a['status'] == 'ambiguous_failure']
check(len(returned) == 3224 and len(failed) == 275, 'returned_and_failed_physical_counts')
actual = sum((a['conservative_usd'] for a in returned), D(0))
uncertain = sum((a['reserved_usd'] for a in failed), D(0))
ledger_cost = sum((r['usd'] for r in ledger if r['event'] == 'call'), D(0))
# The frozen driver accumulates IEEE floats. Decimal sums of stored strings make
# their harmless accumulated rounding explicit instead of silently rounding it.
tolerance = D('0.0000000001')
check(abs(actual - cap['actual_usd']) < tolerance, 'all_returned_costs_retained')
check(abs(uncertain - cap['uncertain_usd']) < tolerance, 'all_failed_reservations_retained_once')
check(abs(actual - ledger_cost) < tolerance, 'returned_and_priced_ledger_costs_match')
check(actual + uncertain < D(8) and cap['total_limit_usd'] == D(20), 'caps_8_and_20_unchanged')
cumulative = D(0)
max_reservation_exposure = D(0)
for a in attempts:
    exposure = cumulative + a['reserved_usd']
    max_reservation_exposure = max(max_reservation_exposure, exposure)
    check(exposure <= D(8) + tolerance, 'each_historical_dispatch_reservation_below_cap', a['attempt'])
    cumulative += a['conservative_usd'] if a['status'] == 'returned' else a['reserved_usd']

cap_denials = [r for r in ledger if r['event'] == 'infra_failure' and r.get('error_kind') == 'smoke_cost_cap']
check(len(cap_denials) == 1, 'one_terminal_cap_denial')
denial = cap_denials[0]
denied_at = timestamp(denial)
check(denial['phase'] == 'screen:162:valid:12:attempt:1', 'cap_denial_last_slot')
post_denial_requests = [r for r in ledger if timestamp(r) > denied_at and r['event'] in ('call', 'infra_retry', 'infra_failure')]
check(not post_denial_requests, 'no_request_events_after_terminal_cap_denial')
check(all(a['started'] is not None and a['started'] < denied_at for a in attempts), 'no_physical_dispatch_after_denial')
check(all(op.get('finished', D(0)) < denied_at for op in ops.values() if op['status'] == 'completed'), 'all_other_episodes_finished_before_denial')
check(set(map(int, screen['tasks'])) == set(range(150, 163)), 'only_ordered_150_through_162_begun')
check(not (RUN / 'operations/screen-162-valid-12-attempt-2.json').exists(), 'no_episode_retry_after_global_cap')
check(not any((RUN / f'task-{task}').exists() for task in range(163, 170)), 'no_later_tasks_dispatched')

last_task = screen['tasks']['162']
last = last_task['episodes'][-1]
check(last_task['status'] == 'INFRA_INCONCLUSIVE' and last_task['executions'] == 12 and last_task['valid_episodes'] == 11 and last_task['valid_successes'] == 0, 'partial_task_is_not_ZERO')
check(last['is_valid'] is False and last['behavioral_success'] is None and last['terminal_errors'] == 1, 'final_error_trace_has_no_behavioral_outcome')
check(last_task['valid_episode_indices'] == list(range(1, 12)) and not (RUN / 'task-162/original-12.json').exists(), 'invalid_execution_excluded_from_valid_indices_and_aliases')
check(ops['screen-162-valid-12-attempt-1.json']['status'] == 'interrupted', 'cap_interrupted_operation_closed')

qualified = {}
check(screen['qualified'] == [154, 158, 159], 'exact_three_qualified_tasks')
for task in screen['qualified']:
    row = screen['tasks'][str(task)]
    check(row['status'] == 'CONFIRMED_ZERO' and row['valid_episodes'] == 16 and row['valid_successes'] == 0 and row['executions'] == 16, 'qualified_exact_0_of_16', task)
    evidence = []
    for index, record in enumerate(row['episodes'], 1):
        original = RUN / record['trace_file']
        raw = read(original)
        alias = RUN / f'task-{task}/original-{index:02d}.json'
        check(read(alias) == raw, 'qualified_valid_alias_identical', str(alias))
        check(record['is_valid'] and record['behavioral_success'] is False and not record['terminal_errors'], 'qualified_valid_behavioral_failure', [task, index])
        traces = json.loads(raw)
        check(len(traces) == 1 and traces[0]['success'] is False and not traces[0].get('error'), 'qualified_trace_completed_failure', [task, index])
        check(traces[0]['rollout_seed'] == task and traces[0]['task_id'] == 'alfworld-corpus-ours-release', 'qualified_trace_bridge_identity', [task, index])
        evidence.append({'valid_index': index, 'episode_id': traces[0]['episode_id'], 'path': str(original.relative_to(ROOT)), 'sha256': hashlib.sha256(raw).hexdigest()})
    qualified[str(task)] = evidence
ids = [r['episode_id'] for rows_ in qualified.values() for r in rows_]
check(len(ids) == len(set(ids)) == 48, '48_unique_qualifying_valid_episodes')
for name in ['screen_audit', 'preservation']:
    path = Path('/tmp/viability_v2_screen_audit_final_independent.json' if name == 'screen_audit' else '/tmp/viability_v2_preservation_final_independent.json')
    prior = load(path)
    check(prior['audit_status'] == 'PASS' and not prior['failures'], name + '_independent_PASS')

# Independently reproduce the seeded selection and every exported trace/hash.
evidence_path = Path('/tmp/viability-v2-screening-evidence/screening_qualified_evidence.json')
exported = load(evidence_path)
check(exported['qualified_task_ids'] == [154, 158, 159], 'export_exact_qualified_membership')
check(exported['adaptation_pool_formed'] is False and exported['LOW_viability_evaluated'] is False and exported['primary_efficacy_denominator'] is None, 'export_no_adaptation_or_efficacy_claim')
check(exported['screening_sha256'] == sources[str((RUN / 'screening.json').relative_to(ROOT))], 'export_terminal_screening_binding')
stage = read(ROOT / 'src/aea/stage.py').decode()
node = next(n for n in ast.parse(stage).body if isinstance(n, ast.FunctionDef) and n.name == 'seeded_failures')
selector_source = ast.get_source_segment(stage, node)
check(hashlib.sha256(stage.encode()).hexdigest() == exported['stage_module_sha256'] == '317b16b7c6de36e41ecc4e07693191b28694ea11e402ff363c340596340b1a28', 'historical_stage_source_frozen')
check(hashlib.sha256(selector_source.encode()).hexdigest() == exported['selection_function_sha256'] == '23020ab07cfe46267fdef6aecb01210275a042f146a6447db9e8b3bef32e65f9', 'historical_seeded_selector_frozen')

def canonical_hash(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True).encode()).hexdigest()

selection_verified = {}
for item in exported['tasks']:
    task = item['task_id']
    originals = [json.loads(read(RUN / f'task-{task}/original-execution-{i:02d}.json'))[0] for i in range(1, 17)]
    chosen = random.Random(task).sample(originals, 3)
    indices = [originals.index(trace) + 1 for trace in chosen]
    check(item['evidence_n'] == 16 and item['selection_n'] == 3 and item['selection_seed'] == task, 'export_counts_and_seed', task)
    check(item['all_original_episode_ids'] == [t['episode_id'] for t in originals], 'export_all16_episode_order', task)
    check(item['all_original_trace_sha256'] == [canonical_hash(t) for t in originals], 'export_all16_canonical_hashes', task)
    check(item['original_zero_evidence_sha256'] == canonical_hash(originals), 'export_all16_collection_hash', task)
    check(item['selected_valid_indices'] == item['selected_execution_indices'] == indices, 'export_exact_seeded_three_order', task)
    check(item['evidence_ids'] == [t['episode_id'] for t in chosen] and item['selected_trace_sha256'] == [canonical_hash(t) for t in chosen], 'export_seeded_three_ids_and_hashes', task)
    check(item['evidence_sha256'] == canonical_hash(chosen), 'export_seeded_three_collection_hash', task)
    check(item['valid_episode_execution_indices'] == list(range(1, 17)), 'export_valid_execution_mapping', task)
    for artifact in item['valid_original_artifacts'] + item['all_execution_artifacts']:
        raw = read(RUN / artifact['path'])
        check(hashlib.sha256(raw).hexdigest() == artifact['file_sha256'], 'export_artifact_file_hash', artifact['path'])
        if 'trace_sha256' in artifact:
            trace = json.loads(raw)[0]
            check(artifact['trace_sha256'] == canonical_hash(trace) and artifact['episode_id'] == trace['episode_id'] and artifact['is_valid'] is True, 'export_execution_metadata', artifact['path'])
    selection_verified[str(task)] = {'all_originals': 16, 'selected_valid_indices': indices, 'evidence_ids': item['evidence_ids']}
last_slot_failed = [a for a in failed if a['attribution']['phase'] == denial['phase']]
last_slot_retries = [r for r in ledger if r['phase'] == denial['phase'] and r['event'] == 'infra_retry']
check(len(last_slot_failed) == len(last_slot_retries) == 1 and last_slot_retries[0]['error_kind'] == 'rate_limit' and last_slot_retries[0]['status_code'] == 429, 'last_slot_prior_physical_429_separate_from_global_cap_denial')
check(timestamp(last_slot_retries[0]) < denied_at and denial['usd'] == D(0), 'cap_denial_creates_no_new_paid_physical_attempt')
prior_failed_artifact = Path('/tmp/viability_v2_terminal_supplement_final.json')
prior_failed_data = load(prior_failed_artifact)
check(len(prior_failed_data['failures']) == 48 and all(f['check'] == 'qualified_trace_bridge_identity' for f in prior_failed_data['failures']), 'initial_supplement_only_schema_mismatch')

result = {
    'schema': 'viability-v2-terminal-cap-and-closure-audit-v2',
    'audit_schema_correction': {'initial_artifact': str(prior_failed_artifact), 'initial_sha256': sources[str(prior_failed_artifact)], 'reason': 'Initial supplemental checker incorrectly compared the trace corpus-label task_id with the numeric bridge ID. Correct schema is task_id=alfworld-corpus-ours-release and rollout_seed=numeric train bridge ID. No experiment or method artifact changed.'},
    'exported_qualified_evidence_verified': selection_verified,
    'last_slot_physical_429': {'physical_attempt_id': last_slot_failed[0]['attempt'], 'reserved_usd_retained': last_slot_failed[0]['reserved_usd'], 'ledger_retry_at': last_slot_retries[0]['ts']},
    'global_cap_denial': {'at': denial['ts'], 'new_physical_attempts': 0, 'new_paid_cost': 0, 'different_event_from_prior_physical_429': True},
    'audit_status': 'FAIL' if failures else 'PASS', 'checks': checks, 'failures': failures,
    'terminal_status': screen['status'], 'cap_denial_at': denial['ts'],
    'last_physical_request_started_unix': max(a['started'] for a in attempts),
    'last_priced_call_at': max(r['ts'] for r in ledger if r['event'] == 'call'),
    'after_denial_events': dict(collections.Counter(r['event'] for r in ledger if timestamp(r) > denied_at)),
    'operation_status_counts': dict(collections.Counter(op['status'] for op in ops.values())),
    'inflight_count': len(cap['inflight']), 'physical_attempts': len(attempts),
    'returned_physical_attempts': len(returned), 'failed_reservations_retained': len(failed),
    'decimal_sums_from_stored_numbers': {'returned': actual, 'failed_reservations': uncertain,
        'committed': actual + uncertain, 'priced_ledger': ledger_cost,
        'cap_actual_rounding_difference': cap['actual_usd'] - actual,
        'cap_uncertain_rounding_difference': cap['uncertain_usd'] - uncertain,
        'maximum_historical_dispatch_reservation_exposure': max_reservation_exposure},
    'qualified_original_traces': qualified,
    'last_partial_task': {'task_id': 162, 'valid_failed_episodes': 11, 'executions': 12,
                         'invalid_final_episode_id': last['episode_ids'][0], 'ZERO': False},
    'source_sha256': sources, 'script_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    'limits': ['No dispatch-start timestamp is stored for operation records; no later operation/attempt, final closed state, and provider/ledger timestamps jointly support the stop claim.',
               'Cap accumulation uses frozen IEEE floats; exact Decimal sums of stored strings expose rounding differences, tested below 1e-10 USD.',
               'This is partial screening evidence. Three qualified tasks do not satisfy the required four-task sample. No efficacy conclusion is established.'],
}
with OUTPUT.open('x') as stream:
    stream.write(json.dumps(result, indent=2, sort_keys=True, default=str) + '\n')
sys.stdout.write(f"{result['audit_status']}: {checks} checks; {len(failures)} failures; committed {actual + uncertain}\n")
if failures:
    sys.stdout.write(json.dumps(failures, indent=2, default=str) + '\n')
raise SystemExit(2 if failures else 0)
