def all_keys(v):
    if isinstance(v,dict):
        for k,x in v.items():
            yield k; yield from all_keys(x)
    elif isinstance(v,list):
        for x in v:yield from all_keys(x)

def independent_changes(before,after,path=()):
    if isinstance(before,dict) and isinstance(after,dict):
        out=[]
        for key in sorted(before.keys() | after.keys()):
            if key in before and key in after:out.extend(independent_changes(before[key],after[key],(*path,key)))
            else:out.append({'path':list((*path,key)),'before_present':key in before,'after_present':key in after,'before':before.get(key),'after':after.get(key)})
        return out
    if isinstance(before,list) and isinstance(after,list) and len(before)==len(after):
        return [c for i,(x,y) in enumerate(zip(before,after)) for c in independent_changes(x,y,(*path,i))]
    if canon(before)==canon(after):return []
    return [{'path':list(path),'before_present':True,'after_present':True,'before':before,'after':after}]

def stream_probes(path):
    rawhash=hashlib.sha256(); canonicalhash=hashlib.sha256(); canonicalhash.update(b'[')
    decoder=json.JSONDecoder(); buffer=''; first=True; started=False; ended=False; count=0
    with gzip.open(path,'rb') as zipped:
        # Input JSON is UTF-8; incremental decoder preserves split code points.
        import codecs
        utf=codecs.getincrementaldecoder('utf-8')()
        while True:
            chunk=zipped.read(262144)
            if chunk:rawhash.update(chunk)
            buffer+=utf.decode(chunk,final=not chunk)
            while True:
                buffer=buffer.lstrip()
                if not started:
                    if not buffer:break
                    if buffer[0]!='[':raise ValueError('expected raw probe array')
                    buffer=buffer[1:];started=True;continue
                if not buffer:break
                if buffer[0]==',':buffer=buffer[1:];continue
                if buffer[0]==']':ended=True;buffer=buffer[1:];break
                try:value,n=decoder.raw_decode(buffer)
                except json.JSONDecodeError:break
                buffer=buffer[n:]
                if not first:canonicalhash.update(b',')
                canonicalhash.update(canon(value).encode());first=False;count+=1
                yield value
            if not chunk:break
        check(ended and not buffer.strip(),'raw_json_stream_complete',str(path))
    canonicalhash.update(b']')
    return rawhash.hexdigest(),canonicalhash.hexdigest(),count
