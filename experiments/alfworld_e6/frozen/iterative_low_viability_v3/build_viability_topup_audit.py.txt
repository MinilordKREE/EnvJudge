def sha(blob): return hashlib.sha256(blob).hexdigest()
def digest(obj): return sha(json.dumps(obj, sort_keys=True).encode())
def git(root, *args):
    return subprocess.check_output(['git', *args], cwd=root, env={**os.environ,'GIT_OPTIONAL_LOCKS':'0'})
def dump(name, obj):
    with (OUT/name).open('x') as stream: json.dump(obj,stream,indent=2,sort_keys=True); stream.write('\n')
def hashpath(path):
    h=hashlib.sha256()
    with path.open('rb') as stream:
        for data in iter(lambda:stream.read(1024*1024),b''):h.update(data)
    return h.hexdigest()

    return {str(p):{'HEAD':git(p,'rev-parse','HEAD').decode().strip(), 'branch':git(p,'branch','--show-current').decode().strip(),'status_short_raw':git(p,'status','--short').decode()} for p in [MAIN,ROOT]}

    record={'worktree':key[0],'path':key[1],'bytes':len(blob),'sha256':checksum,'numeric_fields':{k:sorted(v) for k,v in fields.items()},'path_task_ids':sorted({int(m[1]) for m in PATH_TASK.finditer(key[1])})}
    records.append(record)
    (changed if previous else added).append({'worktree':key[0],'path':key[1],'sha256':checksum,**({'prior_sha256':previous['sha256']} if previous else {})})
